/** schedule records. */
export const schedule = sqliteTable(
  'schedule',
  {
    id: text('id')
      .primaryKey()
      .$defaultFn(() => nanoid(12)),
    ownerType: text('ownerType').notNull(),
    ownerId: text('ownerId').notNull(),
    // Polymorphic owner follows the spec; place IDs are external to Hype.
    purpose: text('purpose', {
      enum: ['openingHours', 'pickupWindow', 'dropoffWindow'],
    })
      .notNull()
      .default('openingHours'),
    osmOpeningHours: text('osmOpeningHours'),
    timezone: text('timezone').notNull().default('Asia/Hong_Kong'),
    isActive: integer('isActive', { mode: 'boolean' }).notNull().default(true),
    createdAt: text('createdAt')
      .notNull()
      .default(sql`(strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))`),
    modifiedAt: text('modifiedAt')
      .notNull()
      .default(sql`(strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))`)
      .$onUpdate(() => new Date().toISOString()),
  },
  table => [
    index('schedule_owner_idx').on(table.ownerType, table.ownerId),
    check('schedule_owner_type', sql`${table.ownerType} IN ('feature','place')`),
  ],
)

/** scheduleRule records. */
export const scheduleRule = sqliteTable(
  'scheduleRule',
  {
    id: text('id')
      .primaryKey()
      .$defaultFn(() => nanoid(12)),
    scheduleId: text('scheduleId')
      .notNull()
      .references(() => schedule.id, { onDelete: 'restrict', onUpdate: 'cascade' }),
    ruleType: text('ruleType', {
      enum: ['weekly', 'nthDoWly', 'publicHoliday'],
    }).notNull(),
    dayOfWeek: integer('dayOfWeek'),
    nth: integer('nth'),
    priority: integer('priority').notNull().default(0),
    validFromDay: integer('validFromDay'),
    validFromMonth: integer('validFromMonth'),
    validFromYear: integer('validFromYear'),
    validUntilDay: integer('validUntilDay'),
    validUntilMonth: integer('validUntilMonth'),
    validUntilYear: integer('validUntilYear'),
    isClosed: integer('isClosed', { mode: 'boolean' }).notNull().default(false),
    is24Hours: integer('is24Hours', { mode: 'boolean' }).notNull().default(false),
    createdAt: text('createdAt')
      .notNull()
      .default(sql`(strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))`),
    modifiedAt: text('modifiedAt')
      .notNull()
      .default(sql`(strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))`)
      .$onUpdate(() => new Date().toISOString()),
  },
  table => [
    index('scheduleRule_schedule_idx').on(table.scheduleId),
    check(
      'scheduleRule_kind',
      sql`${table.ruleType} IN ('weekly','nthDoWly','publicHoliday')`,
    ),
    check(
      'scheduleRule_shape',
      sql`(${table.ruleType} = 'publicHoliday' AND ${table.dayOfWeek} IS NULL AND ${table.nth} IS NULL) OR (${table.ruleType} = 'weekly' AND ${table.dayOfWeek} IS NOT NULL AND ${table.nth} IS NULL) OR (${table.ruleType} = 'nthDoWly' AND ${table.dayOfWeek} IS NOT NULL AND ${table.nth} IS NOT NULL)`,
    ),
    check(
      'scheduleRule_months',
      sql`(${table.validFromMonth} IS NULL OR ${table.validFromMonth} BETWEEN 1 AND 12) AND (${table.validUntilMonth} IS NULL OR ${table.validUntilMonth} BETWEEN 1 AND 12)`,
    ),
    check(
      'scheduleRule_days',
      sql`(${table.validFromDay} IS NULL OR ${table.validFromDay} BETWEEN 1 AND 31) AND (${table.validUntilDay} IS NULL OR ${table.validUntilDay} BETWEEN 1 AND 31)`,
    ),
    check(
      'scheduleRule_open_state',
      sql`NOT (${table.isClosed} AND ${table.is24Hours})`,
    ),
    check('scheduleRule_day', sql`${table.dayOfWeek} BETWEEN 1 AND 7`),
    check('scheduleRule_nth', sql`${table.nth} IN (-1,1,2,3,4,5)`),
  ],
)

/** event records. */
export const event = sqliteTable(
  'event',
  {
    id: text('id')
      .primaryKey()
      .$defaultFn(() => nanoid(12)),
    featureId: text('featureId')
      .notNull()
      .references(() => feature.id, { onDelete: 'restrict', onUpdate: 'cascade' }),
    scheduleId: text('scheduleId').references(() => schedule.id, {
      onDelete: 'restrict',
      onUpdate: 'cascade',
    }),
    type: text('type', {
      enum: ['pickupWindow', 'dropoffWindow', 'programme', 'programmeSlot'],
    }).notNull(),
    fromDate: text('fromDate').notNull(),
    untilDate: text('untilDate').notNull(),
    fromTime: text('fromTime'),
    untilTime: text('untilTime'),
    timezone: text('timezone').notNull().default('Asia/Hong_Kong'),
    // NULL means unlimited; pickup writers explicitly use capacity 1.
    capacity: integer('capacity'),
    atCapacity: integer('atCapacity', { mode: 'boolean' }).notNull().default(false),
    createdAt: text('createdAt')
      .notNull()
      .default(sql`(strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))`),
    modifiedAt: text('modifiedAt')
      .notNull()
      .default(sql`(strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))`)
      .$onUpdate(() => new Date().toISOString()),
  },
  table => [
    index('event_feature_date_idx').on(table.featureId, table.fromDate),
    index('event_date_type_idx').on(table.fromDate, table.type),
    check('event_capacity', sql`${table.capacity} IS NULL OR ${table.capacity} > 0`),
    check('event_dates', sql`${table.untilDate} >= ${table.fromDate}`),
    check(
      'event_kind',
      sql`${table.type} IN ('pickupWindow','dropoffWindow','programme','programmeSlot')`,
    ),
  ],
)

/** Cached route estimates; recompute when event coordinates or order change. */
export const eventTravelLeg = sqliteTable(
  'eventTravelLeg',
  {
    userId: text('userId')
      .notNull()
      .references(() => user.id, { onDelete: 'restrict', onUpdate: 'cascade' }),
    fromEventId: text('fromEventId')
      .notNull()
      .references(() => event.id, { onDelete: 'restrict', onUpdate: 'cascade' }),
    untilEventId: text('untilEventId')
      .notNull()
      .references(() => event.id, { onDelete: 'restrict', onUpdate: 'cascade' }),
    travelMethod: text('travelMethod').notNull(),
    distanceMetres: real('distanceMetres').notNull(),
    durationSeconds: integer('durationSeconds').notNull(),
    createdAt: text('createdAt')
      .notNull()
      .default(sql`(strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))`),
    modifiedAt: text('modifiedAt')
      .notNull()
      .default(sql`(strftime('%Y-%m-%dT%H:%M:%fZ', 'now'))`)
      .$onUpdate(() => new Date().toISOString()),
  },
  table => [
    primaryKey({
      columns: [
        table.userId,
        table.fromEventId,
        table.untilEventId,
        table.travelMethod,
      ],
    }),
    check(
      'eventTravelLeg_metrics',
      sql`${table.distanceMetres} >= 0 AND ${table.durationSeconds} >= 0`,
    ),
  ],
)
